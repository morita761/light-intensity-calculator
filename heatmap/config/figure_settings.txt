# ==========================================================================
# 図と解析パラメータの設定（heatmap/）
#
# [セクション名] = スクリプト名。[common] は全スクリプト共通の既定値。
# 書式は stats_figures/config/figure_settings.txt と同じ:
#   文字列 title = ...（\n で改行）/ 数値 / 組 figsize = 12, 4 / ylim = auto
# タイトル中の {label} は画像セットの label に置き換わる。
# 文字無し版は --no-text / --both で出力（ここの設定は変えなくてよい）
# ==========================================================================

[common]
dpi = 300
format = png
title_fontsize = 12
label_fontsize = 11
tick_fontsize = 10
annotated_title = Annotated {i}
annotated_title_fontsize = 10

# ヒートマップの色
cmap = viridis
vmin = 0
vmax = 255
colorbar_label = Normalized Average GFP Intensity

# --- ROI抽出パラメータ ---
# 馬蹄形をこのサイズ（幅, 高さ）にそろえて平均する
target_size = 100, 100
# この面積(px)未満の輪郭はノイズとして無視
min_area = 20
# マスクの塗り色（blue, cyan, magenta, red から選ぶ）
annotation_colors = blue, cyan, magenta
# 中心線から何px以内の馬蹄形を除外するか（--center-margin で上書き可）
center_margin = 10
# 膨張: バウンディングボックス内の占有率がこの値を超えるまで膨らませる（0 なら膨張しない）
dilate_occupancy = 0.6
# 膨張がこの回数を超えた（細すぎる）馬蹄形は除外
max_dilate = 2
# 明るさ補正後に 0-255 で切り捨てる（false なら小数のまま平均）
clip_uint8 = true
